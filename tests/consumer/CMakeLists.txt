cmake_minimum_required(VERSION 3.20)
project(NocturneConsumer LANGUAGES C CXX)
set(component "consumer-local value")
find_package(Nocturne 1.0 CONFIG REQUIRED COMPONENTS nocturne OPTIONAL_COMPONENTS posix_host windows_host graph spatial simulation)
if(NOT component STREQUAL "consumer-local value")
    message(FATAL_ERROR "Nocturne modified a consumer variable while discovering components")
endif()
enable_testing()
foreach(component IN ITEMS nocturne posix_host windows_host graph spatial simulation)
    if(TARGET Nocturne::${component})
        if(NOT Nocturne_${component}_FOUND OR NOT component IN_LIST Nocturne_INSTALLED_COMPONENTS)
            message(FATAL_ERROR "Installed target ${component} is missing from Nocturne component discovery")
        endif()
    elseif(Nocturne_${component}_FOUND OR component IN_LIST Nocturne_INSTALLED_COMPONENTS)
        message(FATAL_ERROR "Nocturne reports an unavailable component ${component}")
    endif()
endforeach()

add_library(abi_conformance SHARED abi_conformance.c)
set_target_properties(abi_conformance PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON
                      C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_link_libraries(abi_conformance PUBLIC Nocturne::nocturne)
add_executable(abi_conformance_consumer abi_conformance_main.cpp)
set_target_properties(abi_conformance_consumer PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_link_libraries(abi_conformance_consumer PRIVATE abi_conformance)
add_test(NAME abi_conformance_consumer COMMAND abi_conformance_consumer)

add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE Nocturne::nocturne)
add_test(NAME c_consumer COMMAND consumer)
add_executable(cpp_consumer main.cpp)

if(TARGET Nocturne::posix_host)

    if(Python3_Interpreter_FOUND)
        add_executable(posix_host_consumer posix_host.c)
        target_link_libraries(posix_host_consumer PRIVATE Nocturne::posix_host)
        add_test(NAME posix_host_consumer COMMAND "${Python3_EXECUTABLE}"
                 "${CMAKE_CURRENT_SOURCE_DIR}/posix_host.py" $<TARGET_FILE:posix_host_consumer>)
        add_executable(window_terminal_consumer window_terminal.c)
        target_link_libraries(window_terminal_consumer PRIVATE Nocturne::posix_host)
        add_test(NAME window_terminal_consumer COMMAND "${Python3_EXECUTABLE}"
                 "${CMAKE_CURRENT_SOURCE_DIR}/window_terminal.py" $<TARGET_FILE:window_terminal_consumer>)
        add_executable(panel_editor_consumer panel_editor.c)
        target_link_libraries(panel_editor_consumer PRIVATE Nocturne::posix_host)
        add_test(NAME panel_editor_consumer COMMAND "${Python3_EXECUTABLE}"
                 "${CMAKE_CURRENT_SOURCE_DIR}/panel_editor.py" $<TARGET_FILE:panel_editor_consumer>)
        if(TARGET Nocturne::graph)
            add_executable(visual_terminal_consumer visual_terminal.c)
            target_link_libraries(visual_terminal_consumer PRIVATE Nocturne::posix_host Nocturne::graph)
            add_test(NAME visual_terminal_consumer COMMAND "${Python3_EXECUTABLE}"
                     "${CMAKE_CURRENT_SOURCE_DIR}/visual_terminal.py" $<TARGET_FILE:visual_terminal_consumer>)
        endif()
        add_executable(terminal_profile_consumer terminal_profile.c)
        target_link_libraries(terminal_profile_consumer PRIVATE Nocturne::posix_host)
        add_test(NAME terminal_profile_consumer COMMAND "${Python3_EXECUTABLE}"
                 "${CMAKE_CURRENT_SOURCE_DIR}/terminal_profile.py" $<TARGET_FILE:terminal_profile_consumer>)
        find_program(NOCTURNE_TEST_TMUX tmux)
        if(NOCTURNE_TEST_TMUX)
            add_test(NAME terminal_profile_tmux_consumer COMMAND "${Python3_EXECUTABLE}"
                     "${CMAKE_CURRENT_SOURCE_DIR}/terminal_profile.py"
                     $<TARGET_FILE:terminal_profile_consumer> "${NOCTURNE_TEST_TMUX}")
        endif()
    endif()
endif()
